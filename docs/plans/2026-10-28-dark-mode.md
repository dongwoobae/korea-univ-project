**Goal:** 기기 설정 대신 토글 버튼 하나로 공개 지도와 관리자 화면을 다크로 바꾸고(`ku_theme`), 관리자 지도는 늘 밝게 둔다.

**Architecture:** 기준은 `<html data-theme="dark">` 속성이다. `<head>` 인라인 스크립트가 첫 그리기 전에 저장 값을 걸고, 클라이언트
모듈 저장소(`useSyncExternalStore`)가 그 속성을 읽고 바꾼다. 다크 토큰은 `:root[data-theme="dark"]`에, 밝게 둘 지도는
`.ku-light-scope`로 밝은 토큰을 다시 선언한다.

**Tech Stack:** Next 16.2.4(App Router), React 19.2.4(React Compiler), react-leaflet 5, Leaflet 1.9.4, lucide-react, vitest(node
환경), Playwright(목 백엔드 `e2e/support/mockBackend.ts`, `next dev`).

**Spec:** `docs/specs/2026-10-28-dark-mode-design.md`(세부), `docs/specs/2026-10-25-mentoring-feedback-design.md` 7장(요구·사용자 결정).

## Global Constraints

- 저장 키 `ku_theme`. 정확히 `"dark"`일 때만 다크, 없음·`"light"`·모르는 값·예외는 밝음. 쓰기 실패는 화면만 바꾼다(멘토링 설계 2장).
- 기기 설정(`prefers-color-scheme`)을 어디서도 읽지 않는다.
- 토글 이름: 밝을 때 "다크모드"/"Dark mode"/"深色模式", 다크일 때 "라이트모드"/"Light mode"/"浅色模式". 보이는 글자는 영어만
  "Dark"/"Light". `aria-pressed` 없음. 관리자는 한국어만.
- 관리자 지도(작업 지도·폴리곤 편집·미리보기·경사도 경로·관리자 쪽 `FacilityMap`)는 다크에서도 밝다.

- 커밋·주석·문서에 날짜만 적고 과정 이름을 적지 않는다. `Co-Authored-By` 없음.
- 로컬에서 npm ci·vitest·tsc·eslint·Playwright를 돌리지 않는다 — PR CI로 본다. `gh workflow run` 금지. 포맷은 push 전에
  `npx --yes prettier@3.9.4 --write <파일>`.

```ts
import { afterEach, describe, expect, it, vi } from "vitest";
import { THEME_INIT_SCRIPT, parseStoredTheme } from "./theme";
import { setTheme, subscribeTheme } from "./useTheme";

function fakeRoot() {
  const attributes = new Map<string, string>();
  return {
    attributes,
    setAttribute: (name: string, value: string) => attributes.set(name, value),
    removeAttribute: (name: string) => attributes.delete(name),
    getAttribute: (name: string) => attributes.get(name) ?? null,
  };
}

function runInitScript(getItem: (key: string) => string | null) {
  const root = fakeRoot();
  new Function("localStorage", "document", THEME_INIT_SCRIPT)(
    { getItem },
    { documentElement: root },
  );
  return root.attributes.get("data-theme") ?? null;
}

describe("parseStoredTheme", () => {
  it("정확히 dark일 때만 다크다", () => {
    expect(parseStoredTheme("dark")).toBe("dark");
    for (const value of ["light", null, "DARK", "1", ""])
      expect(parseStoredTheme(value)).toBe("light");
  });
});

describe("THEME_INIT_SCRIPT", () => {
  it("저장 값이 dark면 html에 data-theme을 건다", () => {
    expect(runInitScript((key) => (key === "ku_theme" ? "dark" : null))).toBe(
      "dark",
    );
  });

  it("그 밖의 값이면 걸지 않는다", () => {
    for (const value of ["light", null, "DARK"])

  it("저장소가 예외를 던져도 밖으로 내지 않고 밝게 둔다", () => {

        throw new Error("blocked");

  afterEach(() => vi.unstubAllGlobals());

  function stub(setItem: (key: string, value: string) => void) {
    const root = fakeRoot();
    vi.stubGlobal("document", { documentElement: root });
    vi.stubGlobal("localStorage", { setItem });
    return root;
  }

  it("속성과 저장 값을 바꾸고 구독자를 부른다", () => {

    setTheme("dark");
    expect(root.getAttribute("data-theme")).toBe("dark");
    expect(stored.get("ku_theme")).toBe("dark");

    expect(listener).toHaveBeenCalledTimes(2);
  });

  it("저장이 실패해도 화면은 바뀌고 예외를 내지 않는다", () => {
    const root = stub(() => {
      throw new Error("blocked");
    });
    expect(() => setTheme("dark")).not.toThrow();
    expect(root.getAttribute("data-theme")).toBe("dark");
  });
```

```ts
/** 글자를 직접 가진 보이는 요소의 글자색과, 위로 올라가며 처음 만나는 불투명 바탕의 WCAG 대비(설계 5.2) */
async function lowContrast(page: Page, selector: string) {
  return page.evaluate((selector) => {
    type Rgba = { r: number; g: number; b: number; a: number };
    const parse = (value: string): Rgba | null => {
      const rgb = value.match(/^rgba?\(([^)]+)\)$/);
      if (rgb) {
        const [r, g, b, a = 1] = rgb[1]
          .split(/[\s,/]+/)
          .filter(Boolean)
          .map(Number);
        return { r, g, b, a };
      }
      const srgb = value.match(/^color\(srgb ([^)]+)\)$/);
      if (srgb) {
        const [r, g, b, a = 1] = srgb[1]
          .split(/[\s/]+/)
          .filter(Boolean)
          .map(Number);
        return { r: r * 255, g: g * 255, b: b * 255, a };
      }
      return null;
    };
    const luminance = ({ r, g, b }: Rgba) => {
      const channel = (value: number) => {
        const s = value / 255;
        return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
      };
      return 0.2126 * channel(r) + 0.7152 * channel(g) + 0.0722 * channel(b);
    };
    const backdrop = (element: Element): Rgba | null => {
      for (
        let node: Element | null = element;
        node;
        node = node.parentElement
      ) {
        const style = getComputedStyle(node);
        if (style.backgroundImage !== "none") return null;
        const background = parse(style.backgroundColor);
        if (background && background.a >= 0.9) return background;
      }
      return { r: 255, g: 255, b: 255, a: 1 };
    };
    const failures: string[] = [];
    for (const root of Array.from(document.querySelectorAll(selector))) {
      for (const element of [root, ...Array.from(root.querySelectorAll("*"))]) {
        if (
          element.closest(
            ".leaflet-tile-pane, .leaflet-marker-pane, :disabled, [aria-disabled='true']",
          )
        )
          continue;
        const text = Array.from(element.childNodes)
          .filter((node) => node.nodeType === Node.TEXT_NODE)
          .map((node) => node.textContent ?? "")
          .join("")
          .trim();
        if (!text) continue;
        if (
          !element.checkVisibility({
            opacityProperty: true,
            visibilityProperty: true,
          })
        )
          continue;
        const style = getComputedStyle(element);
        const foreground = parse(style.color);
        const background = backdrop(element);
        if (!foreground || !background) continue;
        const mix = (f: number, b: number) =>
          f * foreground.a + b * (1 - foreground.a);
        const shown = {
          r: mix(foreground.r, background.r),
          g: mix(foreground.g, background.g),
          b: mix(foreground.b, background.b),
          a: 1,
        };
        const [high, low] = [luminance(shown), luminance(background)].sort(
          (x, y) => y - x,
        );
        const ratio = (high + 0.05) / (low + 0.05);
        const size = parseFloat(style.fontSize);
        const large =
          size >= 24 || (size >= 18.66 && Number(style.fontWeight) >= 700);
        if (ratio < (large ? 3 : 4.5))
          failures.push(
            `${element.tagName.toLowerCase()}.${element.getAttribute("class") ?? ""} "${text.slice(0, 20)}" ${ratio.toFixed(2)}`,
          );
      }
    }
    return failures;
  }, selector);
}
```

```ts
"use client";

import { useSyncExternalStore } from "react";
import { THEME_STORAGE_KEY, type Theme } from "./theme";

const listeners = new Set<() => void>();

function readTheme(): Theme {
  return document.documentElement.getAttribute("data-theme") === "dark"
    ? "dark"
    : "light";
}

export function subscribeTheme(listener: () => void) {
  listeners.add(listener);
  return () => {
    listeners.delete(listener);
  };
}

export function setTheme(theme: Theme) {
  const root = document.documentElement;
  if (theme === "dark") root.setAttribute("data-theme", "dark");
  else root.removeAttribute("data-theme");
  try {
    localStorage.setItem(THEME_STORAGE_KEY, theme);
  } catch {
    // 저장이 막힌 환경 — 화면만 바꾼다
  }
  for (const listener of listeners) listener();
}

/** 서버와 hydration 동안은 밝음, 그 뒤로는 <html>의 실제 값(설계 2.1) */
export function useTheme(): Theme {
  return useSyncExternalStore(subscribeTheme, readTheme, () => "light");
}
```
